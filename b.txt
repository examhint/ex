import pandas as pd

# --------------------------------------------------
# 1. Create dataset
# --------------------------------------------------

data = {
    "Outlook": [
        "Sunny", "Sunny", "Overcast", "Rain", "Rain",
        "Rain", "Overcast", "Sunny", "Sunny", "Rain"
    ],

    "Temperature": [
        "Hot", "Hot", "Hot", "Mild", "Cool",
        "Cool", "Cool", "Mild", "Cool", "Mild"
    ],

    "Humidity": [
        "High", "High", "High", "High", "Normal",
        "Normal", "Normal", "High", "Normal", "Normal"
    ],

    "Wind": [
        "Weak", "Strong", "Weak", "Weak", "Weak",
        "Strong", "Strong", "Weak", "Weak", "Weak"
    ],

    "Play": [
        "No", "No", "Yes", "Yes", "Yes",
        "No", "Yes", "No", "Yes", "Yes"
    ]
}

df = pd.DataFrame(data)

print("Weather Dataset:")
print(df)


# --------------------------------------------------
# 2. Separate Yes and No
# --------------------------------------------------

yes_data = df[df["Play"] == "Yes"]
no_data = df[df["Play"] == "No"]

yes_count = len(yes_data)
no_count = len(no_data)

print("\nYes records:")
print(yes_data)

print("\nNo records:")
print(no_data)


# --------------------------------------------------
# 3. Prior probabilities
# --------------------------------------------------

total = len(df)

P_yes = yes_count / total
P_no = no_count / total

print("\nPrior Probabilities:")
print("P(Yes) =", P_yes)
print("P(No)  =", P_no)


# --------------------------------------------------
# 4. Given weather condition
# --------------------------------------------------

outlook = "Overcast"
temperature = "Mild"
humidity = "High"
wind = "Strong"


# --------------------------------------------------
# 5. Counts
# --------------------------------------------------

yes_outlook = len(yes_data[yes_data["Outlook"] == outlook])
no_outlook = len(no_data[no_data["Outlook"] == outlook])

yes_temperature = len(
    yes_data[yes_data["Temperature"] == temperature]
)

no_temperature = len(
    no_data[no_data["Temperature"] == temperature]
)

yes_humidity = len(
    yes_data[yes_data["Humidity"] == humidity]
)

no_humidity = len(
    no_data[no_data["Humidity"] == humidity]
)

yes_wind = len(
    yes_data[yes_data["Wind"] == wind]
)

no_wind = len(
    no_data[no_data["Wind"] == wind]
)


print("\nCounts:")
print("Overcast:", yes_outlook, no_outlook)
print("Mild:", yes_temperature, no_temperature)
print("High:", yes_humidity, no_humidity)
print("Strong:", yes_wind, no_wind)


# --------------------------------------------------
# 6. Probabilities WITHOUT Laplace
# --------------------------------------------------

p_outlook_yes = yes_outlook / yes_count
p_outlook_no = no_outlook / no_count

p_temperature_yes = yes_temperature / yes_count
p_temperature_no = no_temperature / no_count

p_humidity_yes = yes_humidity / yes_count
p_humidity_no = no_humidity / no_count

p_wind_yes = yes_wind / yes_count
p_wind_no = no_wind / no_count


print("\nWithout Laplace:")
print("P(Overcast|Yes) =", p_outlook_yes)
print("P(Overcast|No)  =", p_outlook_no)

print("P(Mild|Yes) =", p_temperature_yes)
print("P(Mild|No)  =", p_temperature_no)

print("P(High|Yes) =", p_humidity_yes)
print("P(High|No)  =", p_humidity_no)

print("P(Strong|Yes) =", p_wind_yes)
print("P(Strong|No)  =", p_wind_no)


# --------------------------------------------------
# 7. Scores WITHOUT Laplace
# --------------------------------------------------

score_yes_without = (
    P_yes
    * p_outlook_yes
    * p_temperature_yes
    * p_humidity_yes
    * p_wind_yes
)

score_no_without = (
    P_no
    * p_outlook_no
    * p_temperature_no
    * p_humidity_no
    * p_wind_no
)

print("\nScores WITHOUT Laplace:")
print("Yes =", score_yes_without)
print("No  =", score_no_without)


# --------------------------------------------------
# 8. Laplace Estimation
# --------------------------------------------------

# Number of possible values
k_outlook = df["Outlook"].nunique()
k_temperature = df["Temperature"].nunique()
k_humidity = df["Humidity"].nunique()
k_wind = df["Wind"].nunique()


# Outlook
p_outlook_yes_laplace = (
    yes_outlook + 1
) / (
    yes_count + k_outlook
)

p_outlook_no_laplace = (
    no_outlook + 1
) / (
    no_count + k_outlook
)


# Temperature
p_temperature_yes_laplace = (
    yes_temperature + 1
) / (
    yes_count + k_temperature
)

p_temperature_no_laplace = (
    no_temperature + 1
) / (
    no_count + k_temperature
)


# Humidity
p_humidity_yes_laplace = (
    yes_humidity + 1
) / (
    yes_count + k_humidity
)

p_humidity_no_laplace = (
    no_humidity + 1
) / (
    no_count + k_humidity
)


# Wind
p_wind_yes_laplace = (
    yes_wind + 1
) / (
    yes_count + k_wind
)

p_wind_no_laplace = (
    no_wind + 1
) / (
    no_count + k_wind
)


# --------------------------------------------------
# 9. Display Laplace probabilities
# --------------------------------------------------

print("\nWith Laplace:")
print(
    "P(Overcast|Yes) =",
    p_outlook_yes_laplace
)

print(
    "P(Overcast|No) =",
    p_outlook_no_laplace
)

print(
    "P(Mild|Yes) =",
    p_temperature_yes_laplace
)

print(
    "P(Mild|No) =",
    p_temperature_no_laplace
)

print(
    "P(High|Yes) =",
    p_humidity_yes_laplace
)

print(
    "P(High|No) =",
    p_humidity_no_laplace
)

print(
    "P(Strong|Yes) =",
    p_wind_yes_laplace
)

print(
    "P(Strong|No) =",
    p_wind_no_laplace
)


# --------------------------------------------------
# 10. Scores WITH Laplace
# --------------------------------------------------

score_yes = (
    P_yes
    * p_outlook_yes_laplace
    * p_temperature_yes_laplace
    * p_humidity_yes_laplace
    * p_wind_yes_laplace
)

score_no = (
    P_no
    * p_outlook_no_laplace
    * p_temperature_no_laplace
    * p_humidity_no_laplace
    * p_wind_no_laplace
)


print("\nScores WITH Laplace:")
print("Yes =", score_yes)
print("No  =", score_no)


# --------------------------------------------------
# 11. Prediction
# --------------------------------------------------

if score_yes > score_no:
    print("\nPrediction: Play = Yes")
else:
    print("\nPrediction: Play = No")
