import matplotlib.pyplot as plt
import seaborn as sns


line graph

plt.plot(data["OrderID"], data["Total Sales"], marker="o")
plt.title("Line Chart")
plt.xlabel("Order ID")
plt.ylabel("Total Sales")
plt.grid(True)
plt.show()


bar graph

category_sales = data.groupby("Category")["Total Sales"].sum()

plt.bar(category_sales.index, category_sales.values)
plt.title("Bar Chart")
plt.xlabel("Category")
plt.ylabel("Total Sales")
plt.show()


horizontal bar graph

plt.barh(category_sales.index, category_sales.values)
plt.title("Horizontal Bar Chart")
plt.xlabel("Total Sales")
plt.ylabel("Category")
plt.show()


histogram

plt.hist(data["Total Sales"], bins=8)
plt.title("Histogram")
plt.xlabel("Total Sales")
plt.ylabel("Frequency")
plt.show()


pie chart

category_sales = data.groupby("Category")["Total Sales"].sum()

plt.pie(
    category_sales.values,
    labels=category_sales.index,
    autopct="%1.1f%%"
)

plt.title("Sales Distribution")
plt.show()


scatter plot

plt.scatter(
    data["Quantity"],
    data["Total Sales"]
)

plt.title("Scatter Plot")
plt.xlabel("Quantity")
plt.ylabel("Total Sales")
plt.show()


box plot

plt.boxplot(data["Total Sales"].dropna())

plt.title("Box Plot of Total Sales")
plt.ylabel("Sales")
plt.show()


heatmap

corr = data[
    ["Quantity", "Price", "Total Sales"]
].corr()

plt.imshow(corr, cmap="coolwarm")
plt.colorbar()

plt.xticks(
    range(len(corr.columns)),
    corr.columns,
    rotation=45
)

plt.yticks(
    range(len(corr.columns)),
    corr.columns
)

plt.title("Correlation Heatmap")
plt.tight_layout()
plt.show()
