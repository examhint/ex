# 9. MIN-MAX NORMALIZATION

def min_max_scaling(X):

    minimum = X.min(axis=0)
    maximum = X.max(axis=0)

    return (X - minimum) / (maximum - minimum + 1e-10)


X_minmax = min_max_scaling(X)


# ============================================================
# 10. Z-SCORE NORMALIZATION
# ============================================================

def z_score_scaling(X):

    mean = X.mean(axis=0)
    std = X.std(axis=0)

    return (X - mean) / (std + 1e-10)


X_zscore = z_score_scaling(X)

# ============================================================
# PART B - DISTANCE FUNCTIONS
# ============================================================


# ------------------------------------------------------------
# 1. Euclidean Distance
# ------------------------------------------------------------

def euclidean_distance(x1, x2):

    distance = 0

    for i in range(len(x1)):
        distance += (x1[i] - x2[i]) ** 2

    return np.sqrt(distance)


# ------------------------------------------------------------
# 2. Manhattan Distance
# ------------------------------------------------------------

def manhattan_distance(x1, x2):

    distance = 0

    for i in range(len(x1)):
        distance += abs(x1[i] - x2[i])

    return distance


# ------------------------------------------------------------
# 3. Minkowski Distance
# ------------------------------------------------------------

def minkowski_distance(x1, x2, p=3):

    distance = 0

    for i in range(len(x1)):
        distance += abs(x1[i] - x2[i]) ** p

    return distance ** (1 / p)


# ------------------------------------------------------------
# 4. Cosine Distance
# ------------------------------------------------------------

def cosine_distance(x1, x2):

    dot_product = np.dot(x1, x2)

    magnitude1 = np.sqrt(np.sum(x1 ** 2))
    magnitude2 = np.sqrt(np.sum(x2 ** 2))

    similarity = dot_product / (
        magnitude1 * magnitude2 + 1e-10
    )

    return 1 - similarity


# ------------------------------------------------------------
# 5. Chebyshev Distance
# ------------------------------------------------------------

def chebyshev_distance(x1, x2):

    distance = 0

    for i in range(len(x1)):

        difference = abs(x1[i] - x2[i])

        distance = max(distance, difference)

    return distance


# ============================================================
# PART B - GET DISTANCES TO EVERY TRAINING INSTANCE
# ============================================================

def get_all_distances(X_train, y_train, test_sample, method):

    distances = []

    for i in range(len(X_train)):

        if method == "euclidean":
            dist = euclidean_distance(
                test_sample,
                X_train[i]
            )

        elif method == "manhattan":
            dist = manhattan_distance(
                test_sample,
                X_train[i]
            )

        elif method == "minkowski":
            dist = minkowski_distance(
                test_sample,
                X_train[i]
            )

        elif method == "cosine":
            dist = cosine_distance(
                test_sample,
                X_train[i]
            )

        elif method == "chebyshev":
            dist = chebyshev_distance(
                test_sample,
                X_train[i]
            )

        distances.append((dist, y_train[i]))

    return distances

# ============================================================
# PART C - k-NN FROM SCRATCH
# ============================================================


# ------------------------------------------------------------
# Find k nearest neighbours
# ------------------------------------------------------------

def get_neighbors(
    X_train,
    y_train,
    test_sample,
    k,
    method
):

    distances = []

    for i in range(len(X_train)):

        if method == "euclidean":

            dist = euclidean_distance(
                test_sample,
                X_train[i]
            )

        elif method == "manhattan":

            dist = manhattan_distance(
                test_sample,
                X_train[i]
            )

        elif method == "minkowski":

            dist = minkowski_distance(
                test_sample,
                X_train[i]
            )

        elif method == "cosine":

            dist = cosine_distance(
                test_sample,
                X_train[i]
            )

        elif method == "chebyshev":

            dist = chebyshev_distance(
                test_sample,
                X_train[i]
            )

        distances.append(
            (dist, y_train[i])
        )

    # Sort by distance
    distances.sort(
        key=lambda x: x[0]
    )

    # Select k nearest
    return distances[:k]


# ------------------------------------------------------------
# Majority Voting
# ------------------------------------------------------------

def predict_class(neighbors):

    votes = {}

    for distance, label in neighbors:

        if label not in votes:
            votes[label] = 0

        votes[label] += 1

    return max(
        votes,
        key=votes.get
    )


# ------------------------------------------------------------
# k-NN Prediction
# ------------------------------------------------------------

def knn_predict(
    X_train,
    y_train,
    X_test,
    k,
    method
):

    predictions = []

    for sample in X_test:

        neighbors = get_neighbors(
            X_train,
            y_train,
            sample,
            k,
            method
        )

        prediction = predict_class(
            neighbors
        )

        predictions.append(prediction)

    return predictions


# ============================================================
# ACCURACY
# ============================================================

def accuracy_score(actual, predicted):

    correct = 0

    for i in range(len(actual)):

        if actual[i] == predicted[i]:
            correct += 1

    return (
        correct / len(actual)
    ) * 100

    k,

)

    predictions
)

)


# ============================================================
# PART D - WEIGHTED k-NN
# ============================================================

def weighted_predict(neighbors):

    votes = {}

    for distance, label in neighbors:

        # Weight formula
        weight = 1 / (
            distance + 0.0001
        )

        if label not in votes:
            votes[label] = 0

        votes[label] += weight

    return max(
        votes,
        key=votes.get
    )


def weighted_knn_predict(
    X_train,
    y_train,
    X_test,
    k,
    method
):

    predictions = []

    for sample in X_test:

        neighbors = get_neighbors(
            X_train,
            y_train,
            sample,
            k,
            method
        )

        prediction = weighted_predict(
            neighbors
        )

        predictions.append(prediction)

    return predictions
