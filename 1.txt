cleaning
data = data.drop_duplicates()

num_cols = data.select_dtypes(
    include="number"
).columns

for col in num_cols:
    data[col] = data[col].fillna(
        data[col].mean()
    )

cat_cols = data.select_dtypes(
    include="object"
).columns

for col in cat_cols:
    data[col] = data[col].fillna(
        data[col].mode()[0]
    )


calculate coloumn
data["Total"] = data["Quantity"] * data["Price"]


catagory
def performance(x):
    if x >= 3000:
        return "Excellent"
    elif x >= 1500:
        return "Good"
    else:
        return "Average"

data["Performance"] = data["Total"].apply(
    performance
)

group by
data.groupby("Category")["Total"].agg(
    ["mean", "max", "min"]
)
