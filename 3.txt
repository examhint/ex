min max normalization

def min_max_scaling(series):

    return (
        series - series.min()
    ) / (
        series.max() - series.min()
    )


z score standardization

def z_score_scaling(series):

    return (
        series - series.mean()
    ) / series.std()


euclidean distance

def euclidean_distance(x1, x2):

    distance = 0

    for i in range(len(x1)):

        distance += (
            x1[i] - x2[i]
        ) ** 2

    return np.sqrt(distance)


manhattan distance

def manhattan_distance(x1, x2):

    distance = 0

    for i in range(len(x1)):

        distance += abs(
            x1[i] - x2[i]
        )

    return distance


minkowski distance

def minkowski_distance(
    x1,
    x2,
    p=3
):

    distance = 0

    for i in range(len(x1)):

        distance += (
            abs(x1[i] - x2[i])
        ) ** p

    return distance ** (1/p)


cosine distance

def cosine_distance(x1, x2):

    dot_product = np.dot(
        x1,
        x2
    )

    magnitude1 = np.sqrt(
        np.sum(x1 ** 2)
    )

    magnitude2 = np.sqrt(
        np.sum(x2 ** 2)
    )

    similarity = dot_product / (
        magnitude1 *
        magnitude2 +
        1e-10
    )

    return 1 - similarity


chebyshev distance

def chebyshev_distance(x1, x2):

    distance = 0

    for i in range(len(x1)):

        distance = max(
            distance,
            abs(x1[i] - x2[i])
        )

    return distance

find k nearest neighbours

def get_neighbors(
    X_train,
    y_train,
    test_sample,
    k,
    method
):

    distances = []

    for i in range(
        len(X_train)
    ):

        if method == "euclidean":

            dist = euclidean_distance(
                test_sample,
                X_train[i]
            )

        elif method == "manhattan":

            dist = manhattan_distance(
                test_sample,
                X_train[i]
            )

        elif method == "minkowski":

            dist = minkowski_distance(
                test_sample,
                X_train[i]
            )

        elif method == "cosine":

            dist = cosine_distance(
                test_sample,
                X_train[i]
            )

        elif method == "chebyshev":

            dist = chebyshev_distance(
                test_sample,
                X_train[i]
            )

        distances.append(
            (dist, y_train[i])
        )

    distances.sort(
        key=lambda x: x[0]
    )

    return distances[:k]


majority voting

def predict_class(
    neighbors
):

    votes = {}

    for distance, label in neighbors:

        if label not in votes:

            votes[label] = 0

        votes[label] += 1

    return max(
        votes,
        key=votes.get
    )


k-NN prediction

def knn_predict(
    X_train,
    y_train,
    X_test,
    k,
    method
):

    predictions = []

    for sample in X_test:

        neighbors = get_neighbors(
            X_train,
            y_train,
            sample,
            k,
            method
        )

        prediction = predict_class(
            neighbors
        )

        predictions.append(
            prediction
        )

    return predictions


accuracy

def accuracy_score(
    actual,
    predicted
):

    correct = 0

    for i in range(
        len(actual)
    ):

        if actual[i] == predicted[i]:

            correct += 1

    return (
        correct /
        len(actual)
    ) * 100

weighted k-NN

def weighted_predict(
    neighbors
):

    votes = {}

    for distance, label in neighbors:

        weight = 1 / (
            distance + 0.0001
        )

        if label not in votes:

            votes[label] = 0

        votes[label] += weight

    return max(
        votes,
        key=votes.get
    )
